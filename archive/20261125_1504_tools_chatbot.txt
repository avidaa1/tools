/* ---------------- AI core (visitor-supplied key, direct to Anthropic API) ---------------- */
var AI_KEY='avidaa_anthropic_key';
function aiKey(){try{return localStorage.getItem(AI_KEY)||'';}catch(e){return '';}}
function aiKeyState(){
  var e=document.getElementById('ai-key-state');
  if(e)e.textContent=aiKey()?'KEY SET':'NO KEY';
}
function aiSetKey(){
  var k=(document.getElementById('ai-key-in')||{}).value||'';
  k=k.trim();
  if(!k){setOut('ai-key-out','Paste a key first.','err');return;}
  try{localStorage.setItem(AI_KEY,k);}catch(e){setOut('ai-key-out','Local storage blocked.','err');return;}
  document.getElementById('ai-key-in').value='';
  setOut('ai-key-out','Key stored in this browser only. Never sent anywhere but api.anthropic.com.');
  aiKeyState();
}
function aiClearKey(){
  try{localStorage.removeItem(AI_KEY);}catch(e){}
  setOut('ai-key-out','Key cleared.','warn');
  aiKeyState();
}
function aiCall(prompt,outId,btnId){
  var k=aiKey();
  var btn=btnId?document.getElementById(btnId):null;
  if(!k){
    setOut(outId,'No AI key set. Open Bench > AI Key, paste your own Anthropic API key (stored only in this '+
      'browser, sent only to api.anthropic.com), then try again.','warn');
    return;
  }
  if(btn)btn.disabled=true;
  setOut(outId,'contacting AI...','warn');
  fetch('https://api.anthropic.com/v1/messages',{
    method:'POST',
    headers:{
      'content-type':'application/json',
      'x-api-key':k,
      'anthropic-version':'2023-06-01',
      'anthropic-dangerous-direct-browser-access':'true'
    },
    body:JSON.stringify({model:'claude-sonnet-5',max_tokens:400,messages:[{role:'user',content:prompt}]})
  }).then(function(r){
    if(!r.ok)return r.text().then(function(t){throw new Error('HTTP '+r.status+' '+t.slice(0,160));});
    return r.json();
  }).then(function(d){
    var txt=(d.content||[]).filter(function(b){return b.type==='text';}).map(function(b){return b.text;}).join('\n').trim();
    setOut(outId,txt||'AI returned no text.');
  }).catch(function(e){
    setOut(outId,'AI call failed: '+e.message,'err');
  }).then(function(){if(btn)btn.disabled=false;});
}
